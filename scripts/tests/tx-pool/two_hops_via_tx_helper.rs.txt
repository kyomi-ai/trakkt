pub async fn outer(db: &DbPool) -> trakkt_core::Result<()> {
    let mut tx = db.begin().await?;
    stamp(&mut tx, db).await?; //~ VIOLATION
    tx.commit().await?;
    Ok(())
}

async fn stamp(tx: &mut DbTx, db: &DbPool) -> trakkt_core::Result<()> {
    trakkt_core::tx_execute!(&mut *tx, "UPDATE t SET x = 1")?;
    trakkt_core::db_execute!(db, "UPDATE t SET y = 2")?; //~ VIOLATION
    Ok(())
}
